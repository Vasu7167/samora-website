// ═══════════════════════════════════════════════════════════════════════════ // POST /api/contact — the website enquiry handler. // // WHAT THIS REPLACES // The form posted to https://formspree.io/f/YOUR_FORM_ID, a placeholder that // was never filled in. Every enquiry ever submitted 404'd, the visitor saw the // button change to "Try again", and nothing was recorded anywhere. // // THE ORDERING RULE // Store first, notify second. Email is a notification channel, not a database. // If Resend is down, a key is rotated, or a domain is unverified, we still have // the enquiry and can follow up. The previous design made an email provider a // single point of permanent data loss. // // So: a failed DB write returns an error to the visitor and they can retry. // A failed email does NOT, because the lead is already safe. // // ENV VARS (Vercel → Project → Settings → Environment Variables) // SUPABASE_URL same project as SamoraTrack // SUPABASE_SERVICE_ROLE_KEY server-side only, never shipped to the browser // RESEND_API_KEY // RESEND_FROM e.g. SAM // CONTACT_TO where enquiries land, e.g. vasu@samoraglobal.com // ═══════════════════════════════════════════════════════════════════════════ const SUPABASE_URL = process.env.SUPABASE_URL; const SVC = process.env.SUPABASE_SERVICE_ROLE_KEY; const RESEND_KEY = process.env.RESEND_API_KEY; const MAIL_FROM = process.env.RESEND_FROM || 'SAM '; const CONTACT_TO = process.env.CONTACT_TO || 'vasu@samoraglobal.com'; const esc = (s) => String(s == null ? '' : s) .replace(/&/g, '&').replace(//g, '>') .replace(/"/g, '"'); const clean = (v, max) => String(v == null ? '' : v).trim().slice(0, max); export default async function handler(req, res) { if (req.method !== 'POST') { res.setHeader('Allow', 'POST'); return res.status(405).json({ ok: false, error: 'POST only' }); } // Vercel parses JSON bodies; FormData arrives already decoded too. const body = req.body || {}; // Honeypot. A public endpoint attracts bots within days of going live. // Real people never fill a hidden field, so anything here is automated. // Return 200 so the bot believes it succeeded and does not retry. if (clean(body.website, 200)) { return res.status(200).json({ ok: true }); } const name = clean(body.name, 120); const company = clean(body.company, 160); const email = clean(body.email, 200); const challenge = clean(body.challenge, 200); const message = clean(body.message, 4000); if (!email || !/^[^@\s]+@[^@\s]+\.[^@\s]{2,}$/.test(email)) { return res.status(400).json({ ok: false, error: 'A valid email address is required.' }); } if (!name && !company) { return res.status(400).json({ ok: false, error: 'Tell us your name or your company.' }); } const row = { name: name || null, company: company || null, email, challenge: challenge || null, message: message || null, source: clean(body.source, 80) || 'website', page: clean(req.headers.referer, 300) || null, user_agent: clean(req.headers['user-agent'], 300) || null, }; // ── 1. Store. This one is allowed to fail the request. ────────────────── if (!SUPABASE_URL || !SVC) { console.error('contact: Supabase env vars missing, cannot store enquiry'); return res.status(500).json({ ok: false, error: 'We could not record that. Please email us directly.' }); } let stored = null; try { const r = await fetch(SUPABASE_URL + '/rest/v1/website_enquiries', { method: 'POST', headers: { apikey: SVC, Authorization: 'Bearer ' + SVC, 'Content-Type': 'application/json', Prefer: 'return=representation', }, body: JSON.stringify(row), }); if (!r.ok) { const text = await r.text(); console.error('contact: enquiry insert failed', r.status, text); return res.status(500).json({ ok: false, error: 'We could not record that. Please email us directly.' }); } stored = (await r.json())[0] || null; } catch (e) { console.error('contact: enquiry insert threw', e && e.message); return res.status(500).json({ ok: false, error: 'We could not record that. Please email us directly.' }); } // ── 2. Notify. Failure here is logged, never shown. ───────────────────── // The lead is already saved, so a mail problem must not tell a visitor // their enquiry did not go through. It did. let notified = false; let notifyError = null; if (RESEND_KEY) { try { const html = '

New enquiry from samoraglobal.com

' + '

Name: ' + esc(name || '(not given)') + '

' + '

Company: ' + esc(company || '(not given)') + '

' + '

Email: ' + esc(email) + '

' + '

Challenge: ' + esc(challenge || '(not selected)') + '

' + (message ? '

Message:
' + esc(message).replace(/\n/g, '
') + '

' : '') + '

Page: ' + esc(row.page || 'unknown') + '

'; const text = 'New enquiry from samoraglobal.com\n\n' + 'Name: ' + (name || '(not given)') + '\n' + 'Company: ' + (company || '(not given)') + '\n' + 'Email: ' + email + '\n' + 'Challenge: ' + (challenge || '(not selected)') + '\n' + (message ? '\nMessage:\n' + message + '\n' : ''); const mr = await fetch('https://api.resend.com/emails', { method: 'POST', headers: { Authorization: 'Bearer ' + RESEND_KEY, 'Content-Type': 'application/json' }, body: JSON.stringify({ from: MAIL_FROM, to: [CONTACT_TO], // So a reply goes straight to the person who enquired. reply_to: email, subject: 'Enquiry: ' + (company || name || email), html, text, }), }); const mb = await mr.json().catch(() => ({})); if (!mr.ok) { // The specific mistake that started all of this: the response was // never read, so a rejection looked like a success. notifyError = mb && (mb.message || (mb.error && mb.error.message)) || ('HTTP ' + mr.status); console.error('contact: Resend rejected notification:', notifyError); } else { notified = true; } } catch (e) { notifyError = e && e.message; console.error('contact: Resend call threw', notifyError); } } else { console.error('contact: RESEND_API_KEY not set, enquiry stored but nobody was told'); } // Stored is what matters to the visitor. notified is for our logs. return res.status(200).json({ ok: true, id: stored && stored.id, notified }); }
